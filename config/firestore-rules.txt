rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    function isAdmin() {
      return request.auth != null && request.auth.email == 'ysufrakann@gmail.com';
    }
    
    // Products - Everyone can read, only admin can write
    match /products/{productId} {
      allow read: if true;
      allow write: if isAdmin();
    }
    
    // Users - Users can only read/write their own data
    match /users/{userId} {
      allow read: if request.auth != null && (request.auth.uid == userId || isAdmin());
      allow create: if request.auth != null;
      allow update: if request.auth != null && (request.auth.uid == userId || isAdmin());
      allow delete: if isAdmin();
    }
    
    // Orders - Users can see their own orders, admin can see all
    match /orders/{orderId} {
      allow read: if request.auth != null && (request.auth.uid == resource.data.userId || isAdmin());
      allow create: if request.auth != null;
      allow update: if isAdmin();
      allow delete: if isAdmin();
    }
    
    // Support Requests - Users can create, admin can read/update
    match /support_requests/{requestId} {
      allow read: if isAdmin();
      allow create: if request.auth != null;
      allow update: if isAdmin();
      allow delete: if isAdmin();
    }
    
    // Settings - Everyone can read, only admin can write
    match /settings/{settingId} {
      allow read: if true;
      allow write: if isAdmin();
    }
  }
}
